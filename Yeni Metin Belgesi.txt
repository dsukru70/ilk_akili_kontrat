pragma solidity ^0.8.26;  // "paragma" Hangi Solidity sürümünü kullandığımı belirler 

contract MerhabaDunya { //"contract" Akıllı sözleşme yapısını başlatır
    string public message = "Merhaba Blockchain!";  // "public" Değişkeni veya fonksiyonu dışarıdan görünür yapar yani halka açık
 
    function setMessage(string memory newMessage) public { // "memory" Geçici bellek; işlem sırasında geçerlidir
        message = newMessage; 
    }
}
// "Deploy" Kontratı sanal blockchain’e yükler
// "Cal" Fonksiyonu çalıştırır
